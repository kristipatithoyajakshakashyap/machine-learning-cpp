foreach(t hps gp lin rknn rdtr rrf rgbr rmlp clog cknn csvm cnb cdtc crfc cada cgbc cmlp eval cimb diag ens fsel)
  if(NOT TARGET ${t}_end_to_end OR NOT TARGET ${t}_predict)
    message(STATUS "tests: skipping ${t} (module not registered)")
    continue()
  endif()
  add_test(NAME workflow_${t} COMMAND $<TARGET_FILE:${t}_end_to_end> --quick)
  set_tests_properties(workflow_${t} PROPERTIES LABELS "integration" TIMEOUT 600 FIXTURES_SETUP trained_${t})
  get_target_property(workflow_results ${t}_end_to_end ML_RESULTS_DIR)
  get_target_property(predict_results ${t}_predict ML_RESULTS_DIR)
  add_test(NAME reload_${t} COMMAND ${CMAKE_COMMAND}
    "-DPREDICT=$<TARGET_FILE:${t}_predict>" "-DVERIFY=$<TARGET_FILE:ml_compare_predictions>"
    "-DRESULTS=${workflow_results}" "-DPREDICT_RESULTS=${predict_results}" -P "${CMAKE_CURRENT_SOURCE_DIR}/reload.cmake")
  set_tests_properties(reload_${t} PROPERTIES LABELS "integration;persistence" TIMEOUT 120 FIXTURES_REQUIRED trained_${t})
endforeach()
