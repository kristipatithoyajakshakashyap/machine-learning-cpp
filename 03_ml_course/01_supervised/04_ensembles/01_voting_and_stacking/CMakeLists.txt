# The ensemble is header-only; it links the three base-learner libraries.
set(ML_MODULE_LIB ml_ens)
add_library(${ML_MODULE_LIB} INTERFACE)
target_include_directories(${ML_MODULE_LIB} INTERFACE ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(${ML_MODULE_LIB} INTERFACE ml_clog ml_cknn ml_tree_models)
set(ML_MODULE_RESULTS ${CMAKE_CURRENT_SOURCE_DIR}/results)
add_lesson_executable(ens_theory 01_theory.cpp)
add_lesson_executable(ens_math_intuition 02_math_intuition.cpp)
add_lesson_executable(ens_implementation 03_implementation.cpp)
add_lesson_executable(ens_end_to_end 04_end_to_end.cpp)
add_lesson_executable(ens_predict predict.cpp)
add_lesson_executable(ens_tests tests/model_test.cpp)
add_test(NAME ens_numerical COMMAND ens_tests)
add_test(NAME ens_workflow COMMAND ens_end_to_end --quick)
set_tests_properties(ens_workflow PROPERTIES LABELS "integration")
